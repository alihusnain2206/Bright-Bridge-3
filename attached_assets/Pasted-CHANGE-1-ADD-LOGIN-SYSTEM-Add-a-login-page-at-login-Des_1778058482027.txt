CHANGE 4 — BUILD JWT TOKEN GENERATOR:
Update the existing token endpoint:
POST /api/easyteam/token

Generate DIFFERENT JWT tokens
based on user role:

// Install required package:
// npm install jsonwebtoken

const jwt = require('jsonwebtoken');

app.post('/api/easyteam/token', (req, res) => {
  const { userId } = req.body;
  const user = testUsers.find(u => u.id === userId);
  
  if (!user) {
    return res.status(404).json({ 
      error: 'User not found' 
    });
  }

  // SUPER ADMIN JWT:
  if (user.role === 'super_admin') {
    const payload = {
      employeeId: user.employeeId,
      organizationId: user.companyId,
      partnerId: process.env.EASYTEAM_PARTNER_ID,
      accessRole: {
        name: "admin",
        permissions: [
          "timeclock",
          "timeclock.edit",
          "timesheet.view",
          "timesheet.edit",
          "schedule.view",
          "schedule.manage",
          "timeoff.view",
          "timeoff.manage",
          "reports.view"
        ]
      },
      features: {
        geolocation: false,
        shiftNotes: true
      }
    };
    const token = jwt.sign(
      payload,
      process.env.EASYTEAM_PRIVATE_KEY,
      { algorithm: 'RS256' }
    );
    return res.json({ token, role: 'super_admin' });
  }

  // MANAGER JWT:
  if (user.role === 'manager') {
    const payload = {
      employeeId: user.employeeId,
      organizationId: user.companyId,
      locationId: user.locationId,
      partnerId: process.env.EASYTEAM_PARTNER_ID,
      accessRole: {
        name: "manager",
        permissions: [
          "timeclock",
          "timesheet.view",
          "timesheet.edit",
          "schedule.view",
          "schedule.manage",
          "timeoff.view",
          "timeoff.manage"
        ]
      },
      features: {
        geolocation: false,
        shiftNotes: true
      }
    };
    const token = jwt.sign(
      payload,
      process.env.EASYTEAM_PRIVATE_KEY,
      { algorithm: 'RS256' }
    );
    return res.json({ token, role: 'manager' });
  }

  // EMPLOYEE JWT:
  if (user.role === 'employee') {
    const payload = {
      employeeId: user.employeeId,
      organizationId: user.companyId,
      locationId: user.locationId,
      partnerId: process.env.EASYTEAM_PARTNER_ID,
      role: {
        name: user.position,
        hourlyWage: user.hourlyWage
      },
      accessRole: {
        name: "employee",
        permissions: [
          "timeclock",
          "timesheet.view",
          "timeoff.request",
          "timeoff.view"
        ]
      },
      features: {
        geolocation: true,
        shiftNotes: false
      }
    };
    const token = jwt.sign(
      payload,
      process.env.EASYTEAM_PRIVATE_KEY,
      { algorithm: 'RS256' }
    );
    return res.json({ token, role: 'employee' });
  }

  // PARENT - No EasyTeam access:
  if (user.role === 'parent') {
    return res.json({ 
      token: null,
      role: 'parent',
      message: 'Parents do not have EasyTeam access'
    });
  }
});
