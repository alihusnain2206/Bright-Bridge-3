THE CONFIRMED ROLLFI RESPONSE 
STRUCTURE:

We have confirmed this is exactly
what Rollfi returns from
getPayPeriodDetails:

TOP LEVEL PAY PERIOD:
{
  payPeriod: [{
    payPeriodId: string
    payPeriod: string
    PayPeriodStatus: string
    total: number
    // ← EXACT TOTAL BANK DEBIT!
    // Use this for "Total Bank Debit"
    
    employeeTaxSum: number
    // ← Sum of ALL employee taxes
    
    employerTaxSum: number
    // ← Sum of ALL employer taxes
    
    isProcessed: boolean
    payDate: string
    payBeginDate: string
    payEndDate: string
    workerType: { workerType: string }
    payrollLineItems: [...]
  }]
}

PER EMPLOYEE (payrollLineItems):
{
  userId: string
  // ← Rollfi user ID
  // Use to match with our employee
  
  userName: string
  // ← Employee full name from Rollfi
  
  baseTotal: number
  // ← Base salary/hourly pay
  
  grossTotal: number
  // ← Total gross earnings
  // Includes overtime, bonuses etc
  
  netTotal: number
  // ← NET PAY! Most important!
  // What employee actually receives
  // Use this for "Net Pay" display
  
  payHours: number
  // ← Total hours worked
  
  payRate: number
  // ← Pay rate
  
  retroAmount: number
  // ← Retroactive pay if any
  
  employeeTax: {
    employeeTax: number
    // ← Total employee tax amount
  }
  
  employeeTaxDetails: [
    {
      taxName: string
      // ← EXACT tax label to display!
      // Examples from real Rollfi data:
      // "Federal Withholding Tax"
      // "Medicare"
      // "Social Security"
      // "NJ State Income Tax" (for NJ)
      // "NY State Income Tax" (for NY)
      // "New York City Tax" (for NYC)
      
      taxAmount: number
      // ← EXACT tax amount!
      
      taxAmountYtd: number
      // ← Year to date amount
      
      isEmployerTax: false
      // ← Always false for employee taxes
    }
  ]
  
  employerTaxDetails: [
    {
      taxName: string
      // ← EXACT employer tax label!
      // Examples from real Rollfi data:
      // "FUTA"
      // "Medicare"
      // "Social Security"
      // "NJ State Unemployment Insurance"
      // "NY State Unemployment Insurance"
      
      taxAmount: number
      // ← EXACT employer tax amount!
      
      taxAmountYtd: number
      // ← Year to date amount
      
      isEmployerTax: true
      // ← Always true for employer taxes
    }
  ]
  
  overTimes: [
    {
      type: string
      // ← Overtime type description
      amount: number
      // ← Overtime pay amount
      numberOfHours: number
      // ← Overtime hours
    }
  ]
  
  additionalCompensations: [
    {
      payrollLineItemAdditionalCompensationVertexCompensationIdentifier: {
        compensationDescription: string
        // ← Bonus/comp description
      }
      amount: number
    }
  ]
  
  deductions: [
    {
      vertexDeductionIdentifier: {
        deductionDescription: string
        // ← Deduction description
        // e.g. "401k", "Health Insurance"
      }
      amount: number
    }
  ]
  
  reimbursements: [
    {
      reimbursementType: string
      // ← Type of reimbursement
      amount: number
    }
  ]
  
  payDetails: [
    {
      payPercentage: number
      // ← % going to this account
      amount: number
      // ← Amount to this account
      employeePayAccount: {
        accountName: string
        // ← Bank account description
        // e.g. "savings ... 2500"
      } | null
    }
  ]
}

IMPORTANT TYPESCRIPT RULES:

1. All amounts in dollar format:
   $18.04 not 1804

2. Handle null/undefined safely:
   Use optional chaining:
   payrollLineItem?.netTotal ?? 0
   payrollLineItem?.employeeTaxDetails ?? []

3. Filter zero amounts:
   .filter(tax => tax.taxAmount > 0)
   Never show $0.00 tax lines

4. Keep fromRollfi boolean:
   true = real Rollfi data
   false = estimated fallback

5. TypeScript interfaces:

interface RollfiTaxDetail {
  taxName: string;
  taxAmount: number;
  taxAmountYtd: number;
  isEmployerTax: boolean;
}

interface RollfiPayrollLineItem {
  userId: string;
  userName: string;
  baseTotal: number;
  grossTotal: number;
  netTotal: number;
  payHours: number;
  payRate: number;
  retroAmount: number;
  employeeTax: { employeeTax: number };
  employeeTaxDetails: RollfiTaxDetail[];
  employerTaxDetails: RollfiTaxDetail[];
  overTimes: Array<{
    type: string;
    amount: number;
    numberOfHours: number;
  }>;
  additionalCompensations: Array<{
    payrollLineItemAdditionalCompensationVertexCompensationIdentifier: {
      compensationDescription: string;
    };
    amount: number;
  }>;
  deductions: Array<{
    vertexDeductionIdentifier: {
      deductionDescription: string;
    };
    amount: number;
  }>;
  reimbursements: Array<{
    reimbursementType: string;
    amount: number;
  }>;
  payDetails: Array<{
    payPercentage: number;
    amount: number;
    employeePayAccount: {
      accountName: string;
    } | null;
  }>;
}

interface RollfiPayPeriod {
  payPeriodId: string;
  payPeriod: string;
  PayPeriodStatus: string;
  total: number;
  employeeTaxSum: number;
  employerTaxSum: number;
  isProcessed: boolean;
  payDate: string;
  payBeginDate: string;
  payEndDate: string;
  payrollLineItems: RollfiPayrollLineItem[];
}
