BOTH scenarios must also update
EasyTeam automatically by filtering

ENDPOINT 1 — Deactivate (On Leave):
POST /api/rollfi/employees/deactivate

TypeScript code:
router.post(
'/rollfi/employees/deactivate',
async (req, res) => {
  const { 
    employeeId, 
    reason, 
    expectedReturnDate 
  } = req.body as {
    employeeId: string;
    reason?: string;
    expectedReturnDate?: string;
  };

  // Get employee from store
  const employee = store
    .getAllStaffUsers()
    .find(u => u.employeeId === employeeId);
    
  if (!employee) {
    res.status(404).json({ 
      error: 'Employee not found' 
    });
    return;
  }

  // Get Rollfi mapping
  const rollfiEmp = store
    .getRollfiEmployee(employeeId);
    
  if (!rollfiEmp?.rollfiUserId) {
    res.status(400).json({ 
      error: 'Employee not onboarded 
              to Rollfi' 
    });
    return;
  }

  try {
    // Call Rollfi deactivateUser
    const response = await axios.post(
      `${ROLLFI_BASE_URL}/adminPortal
       #deactivateUser`,
      {
        method: 'deactivateUser',
        userId: rollfiEmp.rollfiUserId
      },
      { headers: rollfiHeaders() }
    );

    req.log.info(
      { rollfiResponse: response.data },
      'Rollfi deactivateUser response'
    );

    // Update employee status in store
    store.updateEmployeeStatus(
      employeeId,
      'on_leave',
      {
        onLeaveReason: reason,
        onLeaveDate: new Date()
          .toISOString(),
        expectedReturnDate
      }
    );

    res.json({ 
      success: true,
      status: 'on_leave',
      rollfiResponse: response.data
    });

  } catch (err: unknown) {
    const e = err as { 
      response?: { data: unknown } 
    };
    req.log.error(
      { err },
      'deactivateUser failed'
    );
    res.status(500).json({ 
      error: 'Failed to deactivate 
              employee',
      details: e.response?.data 
               ?? String(err)
    });
  }
});

ENDPOINT 3 — Reactivate:
POST /api/rollfi/employees/reactivate

router.post(
'/rollfi/employees/reactivate',
async (req, res) => {
  const { employeeId } = req.body as {
    employeeId: string;
  };

  const employee = store
    .getAllStaffUsers()
    .find(u => u.employeeId === employeeId);

  if (!employee) {
    res.status(404).json({ 
      error: 'Employee not found' 
    });
    return;
  }

  // Cannot reactivate terminated!
  if (employee.status === 'terminated') {
    res.status(400).json({ 
      error: 'Cannot reactivate a 
              terminated employee.
              Please add them as 
              a new employee instead.' 
    });
    return;
  }

  const rollfiEmp = store
    .getRollfiEmployee(employeeId);

  if (!rollfiEmp?.rollfiUserId) {
    res.status(400).json({ 
      error: 'Employee not onboarded 
              to Rollfi' 
    });
    return;
  }

  try {
    // Call Rollfi reactivateUser
    const response = await axios.post(
      `${ROLLFI_BASE_URL}/adminPortal
       #reactivateUser`,
      {
        method: 'reactivateUser',
        userId: rollfiEmp.rollfiUserId
      },
      { headers: rollfiHeaders() }
    );

    req.log.info(
      { rollfiResponse: response.data },
      'Rollfi reactivateUser response'
    );

    // Update employee status back to active
    store.updateEmployeeStatus(
      employeeId,
      'active',
      {
        onLeaveReason: undefined,
        onLeaveDate: undefined,
        expectedReturnDate: undefined
      }
    );

    res.json({ 
      success: true,
      status: 'active',
      rollfiResponse: response.data
    });

  } catch (err: unknown) {
    const e = err as { 
      response?: { data: unknown } 
    };
    req.log.error(
      { err },
      'reactivateUser failed'
    );
    res.status(500).json({ 
      error: 'Failed to reactivate 
              employee',
      details: e.response?.data 
               ?? String(err)
    });
  }
});

PART 4 — UPDATE EASYTEAM JWT

This is CRITICAL and very simple!

Find where the EasyTeam JWT
employees array is built.

Currently it probably sends
ALL employees like this:
const employees = 
  store.getAllStaffUsers()
  .filter(u => u.companyId === companyId)

UPDATE to only send ACTIVE employees:
const employees = 
  store.getAllStaffUsers()
  .filter(u => 
    u.companyId === companyId &&
    u.status === 'active'
    // Only active employees
    // can use Time Clock!
    // on_leave and terminated
    // are automatically excluded!
  )

This single filter line means:
→ On leave employees cannot
  clock in via EasyTeam
→ Terminated employees cannot
  clock in via EasyTeam
→ NO separate EasyTeam API call needed!
→ Just filter the array!
